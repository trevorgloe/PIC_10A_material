#include <iostream>

using namespace std;

class A {
private:
  int a1;
  int a2;
  int a3;
  float div_by_2(int &i) { return static_cast<float>(i) / 2.0; }

public:
  A(int a1_, int a2_, int a3_) : a1(a1_), a2(a2_), a3(a3_) {};
  A(float f) : a1(div_by_2(f)), a2(div_by_2(f)), a3(div_by_2(f)) {};
};
int main() {
  A(1, 2, 3);
  A(3);
  return 0;
}
