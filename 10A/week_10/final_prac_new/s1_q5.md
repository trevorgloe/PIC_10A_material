Pay close attention to the scopes in this one...
```cpp
#include <iostream>

using namespace std;

int main() {
  int x = 0;
  int y = 1;
  {
    int &r1 = x;
    int x = 42;
    cout << x << " " << r1 << "\n";
    x *= 2;
  }
  x -= y;
  {
    int y = 100;
    x += y;
  }
  cout << x << " " << y << "\n";
}
```
